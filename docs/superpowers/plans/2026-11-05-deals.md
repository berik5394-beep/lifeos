```ts
import { addCivilDays } from '../../lib/tz.js';

/** Сделки: стадии воронки и follow-up по умолчанию. Чистое (без БД). */
export const DEAL_STAGES = ['new', 'talks', 'proposal', 'contract', 'won', 'lost'] as const;
export type DealStage = (typeof DEAL_STAGES)[number];
export const OPEN_STAGES = ['new', 'talks', 'proposal', 'contract'] as const;
export type OpenStage = (typeof OPEN_STAGES)[number];

export const STAGE_LABEL: Record<DealStage, string> = {
  new: 'Новая',
  talks: 'Переговоры',
  proposal: 'КП отправлено',
  contract: 'Договор/оплата',
  won: 'Выиграна',
  lost: 'Проиграна',
};

const DEFAULT_STEP: Record<OpenStage, { days: number; text: string }> = {
  new: { days: 2, text: 'связаться' },
  talks: { days: 5, text: 'следующий контакт' },
  proposal: { days: 3, text: 'спросить ответ по КП' },
  contract: { days: 3, text: 'проверить оплату' },
};

export function isDealStage(s: string): s is DealStage {
  return (DEAL_STAGES as readonly string[]).includes(s);
}

export function isOpenStage(s: string): s is OpenStage {
  return (OPEN_STAGES as readonly string[]).includes(s);
}

/** Порядок важен: «оплатил» (won) раньше «оплат» (contract), отказ раньше всего. */
const SYNONYMS: Array<[RegExp, DealStage]> = [
  [/(проигр|отказ|слил|не\s+будет|^lost$)/, 'lost'],
  [/(выигр|оплатил|заплатил|купил|^won$)/, 'won'],
  [/(договор|сч[её]т|оплат|^contract$)/, 'contract'],
  [/(^|\s)кп($|\s)|коммерческ|предложени|^proposal$/, 'proposal'],
  [/(перегов|обсужда|общаем|^talks$)/, 'talks'],
  [/(нов|заявк|лид|^new$)/, 'new'],
];

/** Стадия из ключа или русской фразы; не поняли — null (не угадываем). */
export function parseStage(raw: string): DealStage | null {
  const s = raw.trim().toLowerCase().replace(/ё/g, 'е');
  if (!s) return null;
  if (isDealStage(s)) return s;
  for (const [re, stage] of SYNONYMS) if (re.test(s)) return stage;
  return null;
}

/** Следующий шаг по умолчанию при входе в стадию (civil-дата YYYY-MM-DD). */
export function defaultNextStep(stage: DealStage, todayCivil: string): { date: string; text: string } | null {
  if (!isOpenStage(stage)) return null;
  const d = DEFAULT_STEP[stage];
  return { date: addCivilDays(todayCivil, d.days), text: d.text };
}
```

Примечание: в `SYNONYMS` «счёт» нормализуется `ё→е` до проверки, поэтому `сч[её]т` совпадёт.

```ts
/** Темп плана продаж месяца. Чистое. Деньги — в ₸, без сумм сделки не считаются. */
export interface PaceInput {
  plan: number;
  won: number;
  open: number;
  dayOfMonth: number;
  daysInMonth: number;
}
export interface Pace {
  behind: boolean;
  expected: number;
  gap: number;
  coverable: boolean;
}

export function salesPace(p: PaceInput): Pace {
  const expected = p.plan * (p.dayOfMonth / p.daysInMonth);
  const behind = p.dayOfMonth >= 7 && p.won < expected * 0.8;
  const gap = Math.max(0, p.plan - p.won);
  return { behind, expected, gap, coverable: p.open >= gap };
}

/** Крупнейшие открытые сделки, суммарно закрывающие разрыв; не покрыть — пусто. */
export function coveringDeals<T extends { amount: number | null }>(deals: readonly T[], gap: number): T[] {
  if (gap <= 0) return [];
  const sorted = deals
    .filter((d): d is T & { amount: number } => typeof d.amount === 'number' && d.amount > 0)
    .sort((a, b) => b.amount - a.amount);
  const out: T[] = [];
  let sum = 0;
  for (const d of sorted) {
    out.push(d);
    sum += d.amount;
    if (sum >= gap) return out;
  }
  return [];
}
```

```ts
import { ruDayName } from '../goal-slot.js';
import { addCivilDays } from '../../lib/tz.js';
import { STAGE_LABEL, OPEN_STAGES, type DealStage } from './stages.js';

/** Тексты робота про сделки. Чистое: только данные из БД, ничего не додумываем. */

export function fmtMoney(n: number): string {
  if (n >= 1_000_000) {
    const m = n / 1_000_000;
    const s = Number.isInteger(m) ? String(m) : m.toFixed(1).replace('.', ',');
    return `${s} млн ₸`;
  }
  return `${Math.round(n).toString().replace(/\B(?=(\d{3})+(?!\d))/g, ' ')} ₸`;
}

export function whenPhrase(date: string, today: string): string {
  if (date === today) return 'сегодня';
  if (date === addCivilDays(today, 1)) return 'завтра';
  const dd = `${date.slice(8, 10)}.${date.slice(5, 7)}`;
  const day = ruDayName(date);
  const prep = day === 'вторник' ? 'во' : 'в';
  return `${prep} ${accDay(day)} (${dd})`;
}

/** «в среду/пятницу/субботу» — винительный для женского рода дней. */
function accDay(day: string): string {
  return day.endsWith('а') ? day.slice(0, -1) + 'у' : day;
}

/** Родительный «отметил Ахмета», «закроешь Серика»: согласная → +а. */
export function genitive(name: string): string {
  return /[бвгджзклмнпрстфхцчшщ]$/i.test(name) ? `${name}а` : name;
}

interface DealLike {
  personName: string;
  title: string;
  amount: number | null;
  stage: DealStage | string;
}

function head(d: DealLike): string {
  return `${d.personName} — ${d.title}`;
}

export function describeCreated(
  d: DealLike,
  taggedClient: boolean,
  step: { date: string; text: string } | null,
  today: string,
): string {
  const parts = [`Записал сделку: ${head(d)}${d.amount != null ? `, ${fmtMoney(d.amount)}` : ''}, стадия «${STAGE_LABEL[d.stage as DealStage] ?? d.stage}».`];
  if (taggedClient) parts.push(`${genitive(d.personName)} отметил клиентом.`);
  if (step) parts.push(`Напомню ${step.text} ${whenPhrase(step.date, today)}.`);
  return parts.join(' ');
}

export function describeUpdated(
  d: DealLike,
  r: { stageChanged: boolean; step: { date: string; text: string } | null; incomeOffer: boolean },
  today: string,
): string {
  const parts = [`Сделка ${head(d)}: ${r.stageChanged ? `стадия «${STAGE_LABEL[d.stage as DealStage] ?? d.stage}»` : 'обновил'}.`];
  if (r.step) parts.push(`Напомню ${r.step.text} ${whenPhrase(r.step.date, today)}.`);
  if (r.incomeOffer && d.amount != null) parts.push(`Записать доход ${fmtMoney(d.amount)} от ${genitive(d.personName)}?`);
  else if (d.stage === 'won' && d.amount == null) parts.push('Скажи сумму — запишу доход.');
  return parts.join(' ');
}

interface ListDeal extends DealLike {
  nextStepAt: string | null;
  nextStepText: string | null;
}

function dealsWord(n: number): string {
  const m10 = n % 10, m100 = n % 100;
  if (m10 === 1 && m100 !== 11) return 'сделка';
  if (m10 >= 2 && m10 <= 4 && (m100 < 12 || m100 > 14)) return 'сделки';
  return 'сделок';
}

export function describeList(deals: readonly ListDeal[], today: string): string {
  if (deals.length === 0) return 'Открытых сделок нет. Скажи «Ахмет хочет … на 2 млн» — заведу.';
  const sum = deals.reduce((s, d) => s + (d.amount ?? 0), 0);
  const lines = [`Открыто ${deals.length} ${dealsWord(deals.length)}${sum > 0 ? ` на ${fmtMoney(sum)}` : ''}.`];
  for (const st of OPEN_STAGES) {
    const inStage = deals.filter((d) => d.stage === st);
    if (!inStage.length) continue;
    lines.push(`${STAGE_LABEL[st]}: ${inStage.map((d) => `${head(d)}${d.amount != null ? ` (${fmtMoney(d.amount)})` : ''}`).join('; ')}.`);
  }
  const due = deals.filter((d) => d.nextStepAt === today);
  if (due.length) lines.push(`Сегодня: ${due.map((d) => `${d.personName} — ${d.nextStepText ?? 'шаг'}`).join('; ')}.`);
  const overdue = deals.filter((d) => d.nextStepAt != null && d.nextStepAt < today);
  if (overdue.length) lines.push(`Просрочен шаг: ${overdue.map((d) => d.personName).join(', ')}.`);
  return lines.join('\n');
}

export function describePlan(p: {
  plan: number; won: number; open: number; behind: boolean; covering: string[]; noAmount: number;
}): string {
  const parts = [`План месяца ${fmtMoney(p.plan)}: закрыто ${fmtMoney(p.won)}, в работе ещё ${fmtMoney(p.open)}.`];
  if (p.won >= p.plan) parts.push('План выполнен!');
  else if (p.covering.length) parts.push(`Успеешь, если закроешь ${p.covering.map(genitive).join(' и ')}.`);
  else parts.push('В работе не хватает на план — нужны новые сделки.');
  if (p.noAmount > 0) parts.push(`У ${p.noAmount} ${p.noAmount === 1 ? 'сделки' : 'сделок'} нет суммы.`);
  return parts.join(' ');
}

/** Блок для контекста робота (enrichment). Нечего сказать → null. */
export function formatDealsSection(s: {
  openCount: number; openSum: number; dueToday: string[]; overdue: string[]; plan: string | null;
}): string | null {
  if (s.openCount === 0 && !s.plan) return null;
  const parts = [`Сделки: открыто ${s.openCount}${s.openSum > 0 ? ` на ${fmtMoney(s.openSum)}` : ''}`];
  if (s.dueToday.length) parts.push(`шаг сегодня: ${s.dueToday.join(', ')}`);
  if (s.overdue.length) parts.push(`просрочен шаг: ${s.overdue.join(', ')}`);
  if (s.plan) parts.push(s.plan);
  return parts.join('; ') + '.';
}

/** Строка в бриф встречи с клиентом. */
export function dealBriefLine(d: { title: string; amount: number | null; stage: string; daysInStage: number }): string {
  const amount = d.amount != null ? ` ${fmtMoney(d.amount)}` : '';
  return `Сделка «${d.title}»${amount}, ${STAGE_LABEL[d.stage as DealStage] ?? d.stage} ${d.daysInStage} дн назад.`;
}
```

```ts
import { prisma } from '../../lib/prisma.js';
import { getUserTimezone } from '../../lib/user-context.js';
import { localDateStr, localDateOnlyUTC, dateOnlyUTC, localMonthOnlyUTC, localMonthStartUTC } from '../../lib/tz.js';
import { getEntityGraph } from '../entity-graph/index.js';
import { defaultNextStep, isOpenStage, OPEN_STAGES, type DealStage } from './stages.js';
import { salesPace, coveringDeals, type Pace } from './pace.js';

export const DEAL_INCOME_ACTION = 'record_deal_income';

export interface CreateDealInput {
  personName: string;
  title?: string | null;
  amount?: number | null;
  stage?: DealStage;
  nextStepDate?: string | null; // YYYY-MM-DD
  nextStepText?: string | null;
  note?: string | null;
}
export interface UpdateDealPatch {
  stage?: DealStage;
  amount?: number | null;
  nextStepDate?: string | null;
  nextStepText?: string | null;
  note?: string | null;
  lostReason?: string | null;
}
type Step = { date: string; text: string } | null;

const civil = (d: Date | null): string | null => (d ? d.toISOString().slice(0, 10) : null);

async function today(userId: string, now: Date): Promise<{ tz: string; today: string }> {
  const tz = await getUserTimezone(userId);
  return { tz, today: localDateStr(tz, now) };
}

/** Клиент: найти человека (точный тир) или завести; без типа — отметить клиентом. */
async function ensureClient(userId: string, personName: string): Promise<{ entityId: string; tagged: boolean }> {
  const graph = getEntityGraph();
  const found = await graph.resolveForMerge(userId, personName, 'person');
  const attrs = (found?.attributes ?? {}) as Record<string, unknown>;
  if (found && typeof attrs.personType === 'string') return { entityId: found.id, tagged: false };
  const ent = await graph.upsertEntity(
    userId,
    // eslint-disable-next-line @typescript-eslint/no-explicit-any -- Prisma JsonValue (как set_person_type)
    { type: 'person', name: found?.name ?? personName, attributes: { personType: 'client' } as any, importance: 5 },
    { deliberate: true },
  );
  return { entityId: ent.id, tagged: true };
}

export async function createDeal(
  userId: string,
  input: CreateDealInput,
  now: Date = new Date(),
): Promise<{ deal: Deal; taggedClient: boolean; step: Step }> {
  const { today: t } = await today(userId, now);
  const personName = input.personName.trim();
  const { entityId, tagged } = await ensureClient(userId, personName);
  const stage: DealStage = input.stage ?? 'new';
  const step: Step = input.nextStepDate
    ? { date: input.nextStepDate, text: input.nextStepText?.trim() || 'следующий шаг' }
    : defaultNextStep(stage, t);
  const deal = await prisma.deal.create({
    data: {
      userId,
      personEntityId: entityId,
      personName,
      title: input.title?.trim() || 'сделка',
      amount: input.amount ?? null,
      stage,
      nextStepAt: step ? dateOnlyUTC(step.date) : null,
      nextStepText: step?.text ?? null,
      closedAt: isOpenStage(stage) ? null : now,
      events: {
        create: [
          { kind: 'created', to: stage, text: input.note ?? null },
        ],
      },
    },
  });
  return { deal, taggedClient: tagged, step };
}

/** Открытые сделки клиента (+ уточнение названия). Сущность → по FK, иначе по имени. */
export async function findOpenDeals(userId: string, personName: string, title?: string | null): Promise<Deal[]> {
  const ent = await getEntityGraph().resolveForMerge(userId, personName.trim(), 'person');
  const where: Prisma.DealWhereInput = {
    userId,
    stage: { in: [...OPEN_STAGES] },
    OR: [
      ...(ent ? [{ personEntityId: ent.id }] : []),
      { personName: { equals: personName.trim(), mode: 'insensitive' } },
    ],
    ...(title ? { title: { contains: title.trim(), mode: 'insensitive' } } : {}),
  };
  return prisma.deal.findMany({ where, orderBy: { createdAt: 'desc' } });
}

export async function updateDeal(
  userId: string,
  dealId: string,
  patch: UpdateDealPatch,
  now: Date = new Date(),
): Promise<{ deal: Deal; step: Step; stageChanged: boolean; incomeOffer: boolean }> {
  const { today: t } = await today(userId, now);
  const cur = await prisma.deal.findFirst({ where: { id: dealId, userId } });
  if (!cur) throw new Error('not_found');

  const data: Prisma.DealUpdateInput = { lastActivityAt: now };
  const events: Prisma.DealEventCreateWithoutDealInput[] = [];
  let step: Step = null;
  const stageChanged = !!patch.stage && patch.stage !== cur.stage;

  if (stageChanged && patch.stage) {
    data.stage = patch.stage;
    data.stageChangedAt = now;
    events.push({ kind: 'stage', from: cur.stage, to: patch.stage });
    if (isOpenStage(patch.stage)) {
      data.closedAt = null;
      step = defaultNextStep(patch.stage, t);
    } else {
      data.closedAt = now;
      data.nextStepAt = null;
      data.nextStepText = null;
    }
  }
  if (patch.amount !== undefined && patch.amount !== cur.amount) {
    data.amount = patch.amount;
    events.push({ kind: 'amount', from: cur.amount != null ? String(cur.amount) : null, to: patch.amount != null ? String(patch.amount) : null });
  }
  const finalStage = (patch.stage ?? cur.stage) as DealStage;

  const deal = await prisma.deal.update({ where: { id: cur.id }, data });
  if (deal.personEntityId) {
    await prisma.entity.update({ where: { id: deal.personEntityId }, data: { lastSeenAt: now } }).catch(() => {});
  }
  const incomeOffer = stageChanged && deal.stage === 'won' && deal.amount != null && !deal.wonIncomeId;
  return { deal, step, stageChanged, incomeOffer };
}

export type IncomeResult =
  | { ok: true; already: boolean; amount: number }
  | { ok: false; reason: 'not_found' | 'no_amount' | 'not_won' };

/** Доход по выигранной сделке — ровно один раз (wonIncomeId @unique). */
export async function recordDealIncome(userId: string, dealId: string, now: Date = new Date()): Promise<IncomeResult> {
  const deal = await prisma.deal.findFirst({ where: { id: dealId, userId } });
  if (!deal) return { ok: false, reason: 'not_found' };
  if (deal.amount == null) return { ok: false, reason: 'no_amount' };
  if (deal.stage !== 'won') return { ok: false, reason: 'not_won' };
  if (deal.wonIncomeId) return { ok: true, already: true, amount: deal.amount };
  const tz = await getUserTimezone(userId);
  const amount = deal.amount;
  try {
    await prisma.$transaction(async (tx) => {
      const inc = await tx.income.create({
        data: { userId, date: localDateOnlyUTC(tz, now), source: `Сделка: ${deal.personName} — ${deal.title}`, amount },
      });
      const upd = await tx.deal.updateMany({ where: { id: deal.id, wonIncomeId: null }, data: { wonIncomeId: inc.id } });
      if (upd.count === 0) throw new Error('already');
    });
  } catch (err) {
    if (err instanceof Error && err.message === 'already') return { ok: true, already: true, amount };
    throw err;
  }
  return { ok: true, already: false, amount };
}

export async function listDeals(userId: string, status: 'open' | 'closed', now: Date = new Date()): Promise<Deal[]> {
  if (status === 'open') {
    return prisma.deal.findMany({ where: { userId, stage: { in: [...OPEN_STAGES] } }, orderBy: [{ nextStepAt: 'asc' }, { createdAt: 'desc' }] });
  }
  const since = new Date(now.getTime() - 90 * 86_400_000);
  return prisma.deal.findMany({ where: { userId, stage: { in: ['won', 'lost'] }, closedAt: { gte: since } }, orderBy: { closedAt: 'desc' } });
}

export async function setSalesPlan(userId: string, amount: number, now: Date = new Date()): Promise<{ monthStart: Date; amount: number }> {
  const tz = await getUserTimezone(userId);
  const monthStart = localMonthOnlyUTC(tz, now);
  await prisma.salesPlan.upsert({
    where: { userId_monthStart: { userId, monthStart } },
    create: { userId, monthStart, amount },
    update: { amount },
  });
  return { monthStart, amount };
}

export interface PlanStatus {
  plan: number;
  won: number;
  open: number;
  noAmount: number;
  pace: Pace;
  covering: string[];
}

export async function salesPlanStatus(userId: string, now: Date = new Date()): Promise<PlanStatus | null> {
  const tz = await getUserTimezone(userId);
  const monthStart = localMonthOnlyUTC(tz, now);
  const plan = await prisma.salesPlan.findUnique({ where: { userId_monthStart: { userId, monthStart } } });
  if (!plan) return null;
  const monthStartInstant = localMonthStartUTC(tz, now);
  const [wonRows, openRows] = await Promise.all([
    prisma.deal.findMany({ where: { userId, stage: 'won', closedAt: { gte: monthStartInstant } }, select: { amount: true } }),
    prisma.deal.findMany({ where: { userId, stage: { in: [...OPEN_STAGES] } }, select: { personName: true, amount: true } }),
  ]);
  const won = wonRows.reduce((s, d) => s + (d.amount ?? 0), 0);
  const open = openRows.reduce((s, d) => s + (d.amount ?? 0), 0);
  const todayCivil = localDateStr(tz, now);
  const [y, m, d] = todayCivil.split('-').map(Number);
  const daysInMonth = new Date(Date.UTC(y, m, 0)).getUTCDate();
  const pace = salesPace({ plan: plan.amount, won, open, dayOfMonth: d, daysInMonth });
  return {
    plan: plan.amount,
    won,
    open,
    noAmount: openRows.filter((r) => r.amount == null).length,
    pace,
    covering: coveringDeals(openRows, pace.gap).map((r) => r.personName),
  };
}
```

```ts
import type { FastifyInstance, FastifyReply, FastifyRequest } from 'fastify';
import { z } from 'zod';
import { prisma } from '../lib/prisma.js';
import { authMiddleware } from '../middleware/auth.js';
import { validate } from '../middleware/validate.js';
import { isV2DealsEnabled } from '../lib/feature-flags.js';
import {
  createDeal, updateDeal, listDeals, recordDealIncome, setSalesPlan, salesPlanStatus, isDealStage, type DealStage,
} from '../services/deals/index.js';

const civil = z.string().regex(/^\d{4}-\d{2}-\d{2}$/, 'YYYY-MM-DD');
const stage = z.string().refine((s) => isDealStage(s), 'неизвестная стадия');
const createSchema = z.object({
  personName: z.string().min(1).max(120),
  title: z.string().max(200).nullable().optional(),
  amount: z.number().positive().nullable().optional(),
  stage: stage.optional(),
  nextStepDate: civil.nullable().optional(),
  nextStepText: z.string().max(200).nullable().optional(),
});
const updateSchema = z.object({
  stage: stage.optional(),
  amount: z.number().positive().nullable().optional(),
  nextStepDate: civil.nullable().optional(),
  nextStepText: z.string().max(200).nullable().optional(),
  note: z.string().max(500).nullable().optional(),
  lostReason: z.string().max(300).nullable().optional(),
});
const planSchema = z.object({ amount: z.number().positive() });

/** Флаг выключен → 404: мобилка прячет инструмент «Сделки». */
async function dealsGate(request: FastifyRequest, reply: FastifyReply): Promise<FastifyReply | void> {
  // async-хук: после send ОБЯЗАТЕЛЬНО вернуть reply, иначе Fastify пойдёт в хендлер.
  if (!isV2DealsEnabled(request.userId)) return reply.status(404).send({ message: 'Не найдено' });
}

export async function dealRoutes(app: FastifyInstance): Promise<void> {
  app.addHook('preHandler', authMiddleware);
  app.addHook('preHandler', dealsGate);

  app.get('/deals', async (request) => {
    const q = request.query as { status?: string };
    const deals = await listDeals(request.userId, q.status === 'closed' ? 'closed' : 'open');
    return { deals };
  });

  app.get('/deals/:id', async (request, reply) => {
    const { id } = request.params as { id: string };
    const deal = await prisma.deal.findFirst({
      where: { id, userId: request.userId },
      include: { events: { orderBy: { at: 'desc' }, take: 30 } },
    });
    if (!deal) return reply.status(404).send({ message: 'Сделка не найдена' });
    return { deal };
  });

  app.post('/deals', { preHandler: validate(createSchema) }, async (request, reply) => {
    const body = request.body as z.infer<typeof createSchema>;
    const r = await createDeal(request.userId, { ...body, stage: body.stage as DealStage | undefined });
    return reply.status(201).send({ deal: r.deal, step: r.step, taggedClient: r.taggedClient });
  });

  app.put('/deals/:id', { preHandler: validate(updateSchema) }, async (request, reply) => {
    const { id } = request.params as { id: string };
    const body = request.body as z.infer<typeof updateSchema>;
    try {
      const r = await updateDeal(request.userId, id, { ...body, stage: body.stage as DealStage | undefined });
      return { deal: r.deal, step: r.step, incomeOffer: r.incomeOffer };
    } catch (err) {
      if (err instanceof Error && err.message === 'not_found') return reply.status(404).send({ message: 'Сделка не найдена' });
      throw err;
    }
  });

  app.post('/deals/:id/income', async (request, reply) => {
    const { id } = request.params as { id: string };
    const r = await recordDealIncome(request.userId, id);
    if (!r.ok && r.reason === 'not_found') return reply.status(404).send({ message: 'Сделка не найдена' });
    if (!r.ok) return reply.status(422).send({ message: r.reason === 'no_amount' ? 'У сделки нет суммы' : 'Сделка ещё не выиграна' });
    return r;
  });

  app.get('/sales-plan', async (request) => ({ plan: await salesPlanStatus(request.userId) }));

  app.put('/sales-plan', { preHandler: validate(planSchema) }, async (request) => {
    const { amount } = request.body as z.infer<typeof planSchema>;
    await setSalesPlan(request.userId, amount);
    return { plan: await salesPlanStatus(request.userId) };
  });
}
```

```ts
/** Сделки (спека 2026-11-05-deals) — типы и чистая логика панели. Стадии = сервер. */
export const OPEN_STAGES = ['new', 'talks', 'proposal', 'contract'] as const;
export type OpenStage = (typeof OPEN_STAGES)[number];
export type DealStage = OpenStage | 'won' | 'lost';

export const STAGE_LABEL: Record<DealStage, string> = {
  new: 'Новая',
  talks: 'Переговоры',
  proposal: 'КП отправлено',
  contract: 'Договор/оплата',
  won: 'Выиграна',
  lost: 'Проиграна',
};

export interface Deal {
  id: string;
  personName: string;
  title: string;
  amount: number | null;
  stage: DealStage;
  nextStepAt: string | null;
  nextStepText: string | null;
  stageChangedAt: string;
  lastActivityAt: string;
  closedAt: string | null;
  lostReason: string | null;
  wonIncomeId: string | null;
}

export interface DealEvent {
  id: string;
  kind: 'created' | 'stage' | 'amount' | 'next_step' | 'note';
  from: string | null;
  to: string | null;
  text: string | null;
  at: string;
}

export interface PlanStatus {
  plan: number;
  won: number;
  open: number;
  noAmount: number;
  pace: { behind: boolean; expected: number; gap: number; coverable: boolean };
  covering: string[];
}

export interface StageGroup {
  stage: OpenStage;
  deals: Deal[];
  sum: number;
}

export function groupDeals(deals: readonly Deal[]): StageGroup[] {
  return OPEN_STAGES.map((stage) => {
    const inStage = deals.filter((d) => d.stage === stage);
    return { stage, deals: inStage, sum: inStage.reduce((s, d) => s + (d.amount ?? 0), 0) };
  }).filter((g) => g.deals.length > 0);
}

export function neighborStages(stage: OpenStage): { prev: OpenStage | null; next: OpenStage | null } {
  const i = OPEN_STAGES.indexOf(stage);
  return { prev: i > 0 ? OPEN_STAGES[i - 1] : null, next: i < OPEN_STAGES.length - 1 ? OPEN_STAGES[i + 1] : null };
}

export function fmtMoney(n: number): string {
  if (n >= 1_000_000) {
    const m = n / 1_000_000;
    return `${Number.isInteger(m) ? m : m.toFixed(1).replace('.', ',')} млн ₸`;
  }
  return `${Math.round(n).toString().replace(/\B(?=(\d{3})+(?!\d))/g, ' ')} ₸`;
}
```

```ts
import { create } from 'zustand';
import { api, ApiError } from '@/services/api';
import { useAuthStore } from '@/stores/auth-store';
import type { Deal, DealEvent, DealStage, PlanStatus } from '@/services/deals';

interface DealState {
  deals: Deal[];
  plan: PlanStatus | null;
  /** false — сервер ответил 404 (функция выключена) → инструмента нет. */
  available: boolean;
  isLoading: boolean;
  fetch: () => Promise<void>;
  fetchOne: (id: string) => Promise<(Deal & { events: DealEvent[] }) | null>;
  create: (input: { personName: string; title?: string; amount?: number | null }) => Promise<void>;
  update: (id: string, patch: Partial<Pick<Deal, 'amount' | 'nextStepAt' | 'nextStepText' | 'lostReason'>> & { note?: string }) => Promise<void>;
  moveStage: (id: string, stage: DealStage, lostReason?: string) => Promise<{ incomeOffer: boolean }>;
  recordIncome: (id: string) => Promise<{ already: boolean }>;
  setPlan: (amount: number) => Promise<void>;
}

const token = () => useAuthStore.getState().token ?? undefined;

export const useDealStore = create<DealState>((set, get) => ({
  deals: [],
  plan: null,
  available: true,
  isLoading: false,

  fetch: async () => {
    if (!token()) return;
    set({ isLoading: true });
    try {
      const { deals } = await api.get<{ deals: Deal[] }>('/deals?status=open', token());
      const { plan } = await api.get<{ plan: PlanStatus | null }>('/sales-plan', token());
      set({ deals, plan, available: true, isLoading: false });
    } catch (err) {
      set({ isLoading: false, ...(err instanceof ApiError && err.status === 404 ? { available: false } : {}) });
    }
  },

  fetchOne: async (id) => {
    try {
      const { deal } = await api.get<{ deal: Deal & { events: DealEvent[] } }>(`/deals/${id}`, token());
      return deal;
    } catch {
      return null;
    }
  },

  create: async (input) => {
    await api.post('/deals', input, token());
    await get().fetch();
  },

  update: async (id, patch) => {
    const { deal } = await api.put<{ deal: Deal }>(`/deals/${id}`, patch, token());
    set((s) => ({ deals: s.deals.map((d) => (d.id === id ? deal : d)) }));
  },

  moveStage: async (id, stage, lostReason) => {
    const before = get().deals;
    set((s) => ({ deals: s.deals.map((d) => (d.id === id ? { ...d, stage } : d)) }));
    try {
      const r = await api.put<{ deal: Deal; incomeOffer: boolean }>(`/deals/${id}`, { stage, ...(lostReason ? { lostReason } : {}) }, token());
      set((s) => ({
        deals: r.deal.stage === 'won' || r.deal.stage === 'lost' ? s.deals.filter((d) => d.id !== id) : s.deals.map((d) => (d.id === id ? r.deal : d)),
      }));
      const { plan } = await api.get<{ plan: PlanStatus | null }>('/sales-plan', token());
      set({ plan });
      return { incomeOffer: r.incomeOffer };
    } catch (err) {
      set({ deals: before });
      throw err;
    }
  },

  recordIncome: async (id) => {
    const r = await api.post<{ ok: true; already: boolean }>(`/deals/${id}/income`, {}, token());
    return { already: r.already };
  },

  setPlan: async (amount) => {
    const { plan } = await api.put<{ plan: PlanStatus | null }>('/sales-plan', { amount }, token());
    set({ plan });
  },
}));
```

```tsx
import React, { memo, useCallback, useEffect, useState } from 'react';
import { Alert, Modal, Pressable, ScrollView, StyleSheet, Text, View } from 'react-native';
import { useColors } from '@/hooks/use-colors';
import { spacing, fontSize, borderRadius } from '@/constants';
import { STAGE_LABEL, neighborStages, fmtMoney, type Deal, type DealEvent, type OpenStage } from '@/services/deals';
import { useDealStore } from '@/stores/deal-store';

export const DealSheet = memo(function DealSheet({ deal, onClose }: { deal: Deal | null; onClose: () => void }) {
  const c = useColors();
  const moveStage = useDealStore((s) => s.moveStage);
  const recordIncome = useDealStore((s) => s.recordIncome);
  const fetchOne = useDealStore((s) => s.fetchOne);
  const [events, setEvents] = useState<DealEvent[]>([]);

  useEffect(() => {
    setEvents([]);
    if (deal) void fetchOne(deal.id).then((d) => setEvents(d?.events ?? []));
  }, [deal, fetchOne]);

  const onMove = useCallback(
    async (stage: OpenStage | 'won' | 'lost', lostReason?: string) => {
      if (!deal) return;
      try {
        const r = await moveStage(deal.id, stage, lostReason);
        if (stage === 'won' && r.incomeOffer && deal.amount != null) {
          Alert.alert('Сделка выиграна!', `Записать доход ${fmtMoney(deal.amount)} от ${deal.personName}?`, [
            { text: 'Не сейчас', style: 'cancel', onPress: onClose },
            { text: 'Записать', onPress: () => void recordIncome(deal.id).finally(onClose) },
          ]);
          return;
        }
        onClose();
      } catch {
        Alert.alert('Не получилось', 'Проверь интернет и попробуй ещё раз.');
      }
    },
    [deal, moveStage, recordIncome, onClose],
  );

  const onLost = useCallback(() => {
    Alert.alert('Почему проиграна?', undefined, [
      { text: 'Дорого', onPress: () => void onMove('lost', 'дорого') },
      { text: 'Выбрал другого', onPress: () => void onMove('lost', 'выбрал другого') },
      { text: 'Без причины', onPress: () => void onMove('lost') },
      { text: 'Отмена', style: 'cancel' },
    ]);
  }, [onMove]);

  if (!deal) return null;
  const open = deal.stage !== 'won' && deal.stage !== 'lost';
  const nb = open ? neighborStages(deal.stage as OpenStage) : { prev: null, next: null };

  return (
    <Modal visible transparent animationType="slide" onRequestClose={onClose}>
      <Pressable style={styles.backdrop} onPress={onClose} />
      <View style={[styles.sheet, { backgroundColor: c.background }]}>
        <Text style={[styles.title, { color: c.text }]}>{`${deal.personName} — ${deal.title}`}</Text>
        <Text style={[styles.meta, { color: c.textSecondary }]}>
          {`${STAGE_LABEL[deal.stage]}${deal.amount != null ? ` · ${fmtMoney(deal.amount)}` : ' · сумма не названа'}`}
        </Text>
        {deal.nextStepText ? <Text style={[styles.meta, { color: c.textSecondary }]}>{`Шаг: ${deal.nextStepText}`}</Text> : null}
        <View style={styles.row}>
          {nb.prev ? <Btn label={`← ${STAGE_LABEL[nb.prev]}`} onPress={() => void onMove(nb.prev!)} /> : null}
          {nb.next ? <Btn label={`${STAGE_LABEL[nb.next]} →`} onPress={() => void onMove(nb.next!)} /> : null}
        </View>
        <View style={styles.row}>
          <Btn label="Выиграна" color={c.success} onPress={() => void onMove('won')} />
          <Btn label="Проиграна" color={c.danger} onPress={onLost} />
        </View>
        <ScrollView style={styles.history}>
          {events.map((e) => (
            <Text key={e.id} style={[styles.event, { color: c.textSecondary }]}>
              {`${e.at.slice(8, 10)}.${e.at.slice(5, 7)} · ${eventText(e)}`}
            </Text>
          ))}
        </ScrollView>
      </View>
    </Modal>
  );
});

function eventText(e: DealEvent): string {
  if (e.kind === 'created') return 'сделка заведена';
  if (e.kind === 'stage') return `стадия: ${STAGE_LABEL[(e.to ?? 'new') as keyof typeof STAGE_LABEL] ?? e.to}`;
  if (e.kind === 'amount') return `сумма: ${e.to ? fmtMoney(Number(e.to)) : '—'}`;
  if (e.kind === 'next_step') return `шаг: ${e.text ?? ''}`;
  return e.text ?? '';
}

function Btn({ label, onPress, color }: { label: string; onPress: () => void; color?: string }) {
  const c = useColors();
  return (
    <Pressable onPress={onPress} style={[styles.btn, { backgroundColor: color ?? c.surface }]} accessibilityRole="button">
      <Text style={[styles.btnText, { color: color ? '#FFFFFF' : c.text }]}>{label}</Text>
    </Pressable>
  );
}

const styles = StyleSheet.create({
  backdrop: { flex: 1, backgroundColor: 'rgba(0,0,0,0.4)' },
  sheet: { padding: spacing.lg, borderTopLeftRadius: borderRadius.xl, borderTopRightRadius: borderRadius.xl, maxHeight: '70%' },
  title: { fontSize: fontSize.lg, fontWeight: '700' },
  meta: { fontSize: fontSize.sm, marginTop: spacing.xs },
  row: { flexDirection: 'row', gap: spacing.sm, marginTop: spacing.md },
  btn: { flex: 1, paddingVertical: spacing.sm + 2, borderRadius: borderRadius.md, alignItems: 'center' },
  btnText: { fontSize: fontSize.sm, fontWeight: '600' },
  history: { marginTop: spacing.md },
  event: { fontSize: fontSize.xs, marginBottom: 4 },
});
```

```tsx
};

export default function DealsScreen() {
  const c = useColors();
  const deals = useDealStore((s) => s.deals);
```
